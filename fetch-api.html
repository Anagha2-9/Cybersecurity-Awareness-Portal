How Fetch API Works

Modern asynchronous pattern using async/await: 

async function loadTips() {
  try {
    const res = await fetch('data/tips.json');
    if (!res.ok) throw new Error(res.status);
    const data = await res.json();
    renderData(data);
  } catch (error) {
    console.error('Fetch error:', error);
  }
}
